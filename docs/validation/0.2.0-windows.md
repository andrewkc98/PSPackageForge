Import-Module C:\PSPackageForge-Acceptance\repo-task29-7a5f164\PSPackageForge.psd1 -Force

        [EvidenceRecord]::new('InstallCommand', @{ Executable = 'unknown.exe'; ArgumentList = @('/quiet'); ExpectedExitCodes = @(0) }, [EvidenceSource]::UserOverride, [ConfidenceLevel]::High),
        [EvidenceRecord]::new('UninstallCommand', @{ Executable = 'uninstall.exe'; ArgumentList = @('/quiet'); ExpectedExitCodes = @(0) }, [EvidenceSource]::UserOverride, [ConfidenceLevel]::High),
        [EvidenceRecord]::new('DetectionTarget', 'C:\Program Files\Reviewed\unknown.exe', [EvidenceSource]::UserOverride, [ConfidenceLevel]::High)
    )
    $info = Get-InstallerInfo -Path $path -AdditionalEvidence $evidence
    $spec = Resolve-PackageSpec -InstallerInfo $info
    [pscustomobject]@{ Readiness = $spec.Readiness.ToString(); Warnings = [string]::Join(';', @($info.Findings | ForEach-Object { $_.ToString() })); BlockingCount = @($spec.BlockingFindings).Count; Framework = @($info.ResolvedEvidence | Where-Object Field -eq 'Framework' | ForEach-Object Value) }
